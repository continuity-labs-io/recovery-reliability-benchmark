@frozen('SPEC min-dataset-3')
def check_scored_units(scored: Mapping[str, Mapping[str, Set[str]]], measure: str, condition: str) -> None:
    n_scored = len(scored.get(measure, {}).get(condition, set()))
    if n_scored < contract.MD3_MUST:
        msg = f"{n_scored} units with {measure} computed in both halves for condition '{condition}', requires {contract.MD3_MUST}."
        raise ContractViolation(Requirement.MD3, msg, found=n_scored, required=contract.MD3_MUST)
